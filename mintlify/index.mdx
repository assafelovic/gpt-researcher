export const CopyInstall = () => {
  const [copied, setCopied] = useState(false);
  const copy = async () => {
    try {
      await navigator.clipboard.writeText(INSTALL);
      setCopied(true);
      setTimeout(() => setCopied(false), 1600);
    } catch (e) {}
  };
  return (
    <button type="button" className="gptr-install" onClick={copy} aria-label="Copy install command">
      <span className="gptr-prompt">$</span>
      <code>{INSTALL}</code>
      <span className="gptr-copy" aria-live="polite">{copied ? "Copied" : "Copy"}</span>
    </button>
  );
};
